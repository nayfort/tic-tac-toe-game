cmake_minimum_required(VERSION 3.16)
project(TicTacToe VERSION 1.0.0 LANGUAGES CXX)

find_package(OpenGL REQUIRED)
find_package(GLUT REQUIRED)
add_library(game Board.cpp Cross.cpp Disk.cpp Scene.cpp Shape.cpp utils.cpp)
target_compile_features(game PUBLIC cxx_std_17)
target_include_directories(game PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(game PUBLIC OpenGL::GL OpenGL::GLU GLUT::GLUT)
if(MSVC)
    target_compile_options(game PUBLIC /W4 /utf-8)
else()
    target_compile_options(game PUBLIC -Wall -Wextra -Wpedantic)
endif()
add_executable(tic-tac-toe main.cpp)
target_link_libraries(tic-tac-toe PRIVATE game)

include(CTest)
if(BUILD_TESTING)
    add_executable(game-tests tests/game_tests.cpp)
    target_link_libraries(game-tests PRIVATE game)
    add_test(NAME game-regressions COMMAND game-tests)
endif()
